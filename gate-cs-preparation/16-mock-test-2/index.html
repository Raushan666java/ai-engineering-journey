The \GATEScoreCalculator\ class (enhanced) adds multi-section score tracking and aggregate analysis for mock test series.

\\ ypescript /**

  • GATEScoreCalculator — Enhanced version with
  • multi-section tracking and aggregate analysis. */ interface SectionResult { name: string; correct: number; wrong: number; attempted: number; }

class GATEScoreCalculator { private sections: Map<string, { marks: number; penalty: number }>;

constructor() { this.sections = new Map(); }

/** Register a section with its marking scheme. */ addSection(name: string, marksPerQ: number, penaltyPerQ: number): void { this.sections.set(name, { marks: marksPerQ, penalty: penaltyPerQ }); }

/** Compute per-section raw score. */ sectionRaw(section: string, correct: number, wrong: number): number { const config = this.sections.get(section); if (!config) throw new Error(\Unknown section: \); return correct * config.marks - wrong * config.penalty; }

/** Compute total raw score from multiple section results. */ totalRaw(results: SectionResult[]): number { let total = 0; for (const r of results) { total += this.sectionRaw(r.name, r.correct, r.wrong); } return Math.round(total * 100) / 100; }

/** Aggregate multiple mock test scores into a report. */ aggregateReport(allScores: number[][]): { averages: Map<string, number>; overall: number; trend: string; } { const averages = new Map<string, number>(); let grandTotal = 0; let count = 0; for (const test of allScores) { for (let i = 0; i < test.length; i++) { const key = \Section-\; const prev = averages.get(key) ?? 0; averages.set(key, prev + test[i]); } grandTotal += test.reduce((a, b) => a + b, 0); count++; } for (const [k, v] of averages) { averages.set(k, Math.round((v / count) * 100) / 100); } const trend = this.determineTrend(allScores); return { averages, overall: Math.round((grandTotal / count) * 100) / 100, trend }; }

private determineTrend(scores: number[][]): string { const totals = scores.map((s) => s.reduce((a, b) => a + b, 0)); if (totals.length < 2) return 'Insufficient data'; const improving = totals[totals.length - 1] > totals[0]; return improving ? 'Improving ?' : 'Declining ?'; } }
