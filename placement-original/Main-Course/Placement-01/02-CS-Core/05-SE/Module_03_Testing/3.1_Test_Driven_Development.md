# Test-Driven Development (TDD) - Complete Implementation Guide
## Writing Tests First, Code Second

## 📋 Overview
Test-Driven Development (TDD) is a software development methodology where tests are written before the actual code. This approach ensures that code is thoroughly tested, well-designed, and meets requirements from the start.

## 🔄 The TDD Cycle: Red-Green-Refactor

### Red Phase: Write a Failing Test
```python
# Example: Calculator Addition Function
import unittest

class TestCalculator(unittest.TestCase):
    def setUp(self):
        # This will fail initially because Calculator doesn't exist
        self.calculator = Calculator()
    
    def test_add_two_positive_numbers(self):
        # Red: This test will fail because add method doesn't exist
        result = self.calculator.add(2, 3)
        self.assertEqual(result, 5)
    
    def test_add_positive_and_negative_number(self):
        result = self.calculator.add(5, -3)
        self.assertEqual(result, 2)
    
    def test_add_two_negative_numbers(self):
        result = self.calculator.add(-2, -3)
        self.assertEqual(result, -5)
    
    def test_add_zero_to_number(self):
        result = self.calculator.add(5, 0)
        self.assertEqual(result, 5)

if __name__ == '__main__':
    unittest.main()
```

### Green Phase: Write Minimal Code to Pass
```python
# calculator.py - Minimal implementation to make tests pass
class Calculator:
    def add(self, a, b):
        return a + b  # Simplest implementation that works

# Run tests - they should now pass (Green)
```

### Refactor Phase: Improve Code Quality
```python
# calculator.py - Refactored version with better design
class Calculator:
    """A simple calculator class for basic arithmetic operations."""
    
    def add(self, a, b):
        """
        Add two numbers and return the result.
        
        Args:
            a (int/float): First number
            b (int/float): Second number
            
        Returns:
            int/float: Sum of a and b
            
        Raises:
            TypeError: If inputs are not numbers
        """
        if not isinstance(a, (int, float)) or not isinstance(b, (int, float)):
            raise TypeError("Both arguments must be numbers")
        
        return a + b
```

### 1. Triangulation Pattern
```python
class TestStringCalculator(unittest.TestCase):
    def setUp(self):
        self.calculator = StringCalculator()
    
    def test_empty_string_returns_zero(self):
        # First test - specific case
        result = self.calculator.add("")
        self.assertEqual(result, 0)
    
    def test_single_number_returns_that_number(self):
        # Second test - another specific case
        result = self.calculator.add("1")
        self.assertEqual(result, 1)
        
        # Triangulate with different value
        result = self.calculator.add("5")
        self.assertEqual(result, 5)
    
    def test_two_numbers_comma_delimited_returns_sum(self):
        # Third test - forces generalization
        result = self.calculator.add("1,2")
        self.assertEqual(result, 3)
        
        # Triangulate
        result = self.calculator.add("3,4")
        self.assertEqual(result, 7)

class StringCalculator:
    def add(self, numbers):
        if not numbers:
            return 0
        
        if ',' not in numbers:
            return int(numbers)
        
        # Split by comma and sum
        return sum(int(num) for num in numbers.split(','))
```

import requests

        }
        mock_response.status_code = 200
        mock_get.return_value = mock_response
        
        # Act
        temperature = self.weather_service.get_temperature('London')
        
        # Assert
        self.assertEqual(temperature, 25.5)
        mock_get.assert_called_once_with(
            'http://api.openweathermap.org/data/2.5/weather',

        with self.assertRaises(WeatherServiceError):

    pass

class InsufficientFundsError(Exception):
    pass

class BankAccount:
    def __init__(self):
        self._balance = 0
    
    @property
    def balance(self):
        return self._balance
    
    def deposit(self, amount):
        if amount < 0:
            raise ValueError("Deposit amount must be positive")
        self._balance += amount
    
    def withdraw(self, amount):
        if amount < 0:
            raise ValueError("Withdrawal amount must be positive")
        if amount > self._balance:
            raise InsufficientFundsError("Insufficient funds")
        self._balance -= amount

### Code Coverage Analysis
```python
# coverage_analyzer.py
import coverage
import unittest

class TDDMetrics:
    def __init__(self):
        self.cov = coverage.Coverage()
        self.test_results = {}
    
    def run_tests_with_coverage(self, test_module):
        """Run tests and measure code coverage"""
        self.cov.start()
        
        # Run tests
        loader = unittest.TestLoader()
        suite = loader.loadTestsFromModule(test_module)
        runner = unittest.TextTestRunner(verbosity=2)
        result = runner.run(suite)
        
        self.cov.stop()
        self.cov.save()
        
        # Generate coverage report
        coverage_report = self.cov.report()
        
        return {
            'tests_run': result.testsRun,
            'failures': len(result.failures),
            'errors': len(result.errors),
            'success_rate': (result.testsRun - len(result.failures) - len(result.errors)) / result.testsRun * 100,
            'coverage_percentage': coverage_report
        }
    
    def analyze_test_quality(self, test_cases):
        """Analyze test quality metrics"""
        metrics = {
            'total_tests': len(test_cases),
            'assertion_count': 0,
            'test_complexity': [],
            'naming_quality': []
        }
        
        for test_case in test_cases:
            # Count assertions (simplified)
            test_code = inspect.getsource(test_case)
            assertion_count = test_code.count('assert')
            metrics['assertion_count'] += assertion_count
            
            # Analyze test complexity (lines of code)
            complexity = len(test_code.split('\n'))
            metrics['test_complexity'].append(complexity)
            
            # Check naming quality
            name_quality = self._assess_test_name_quality(test_case.__name__)
            metrics['naming_quality'].append(name_quality)
        
        return metrics
    
    def _assess_test_name_quality(self, test_name):
        """Assess quality of test name"""
        quality_score = 0
        
        # Check if name starts with 'test_'
        if test_name.startswith('test_'):
            quality_score += 1
        
        # Check if name is descriptive (contains action words)
        action_words = ['should', 'when', 'given', 'returns', 'throws', 'creates']
        if any(word in test_name.lower() for word in action_words):
            quality_score += 1
        
        # Check length (not too short, not too long)
        if 20 <= len(test_name) <= 80:
            quality_score += 1
        
        return quality_score / 3  # Normalize to 0-1 scale
```

### TDD Cycle Time Tracking
```python
import time
from datetime import datetime, timedelta

class TDDCycleTracker:
    def __init__(self):
        self.cycles = []
        self.current_cycle = None
    
    def start_red_phase(self, test_description):
        """Start timing the Red phase"""
        self.current_cycle = {
            'test_description': test_description,
            'red_start': datetime.now(),
            'phases': {}
        }
    
    def start_green_phase(self):
        """Transition from Red to Green phase"""
        if self.current_cycle:
            now = datetime.now()
            self.current_cycle['phases']['red_duration'] = (
                now - self.current_cycle['red_start']
            ).total_seconds()
            self.current_cycle['green_start'] = now
    
    def start_refactor_phase(self):
        """Transition from Green to Refactor phase"""
        if self.current_cycle:
            now = datetime.now()
            self.current_cycle['phases']['green_duration'] = (
                now - self.current_cycle['green_start']
            ).total_seconds()
            self.current_cycle['refactor_start'] = now
    
    def complete_cycle(self):
        """Complete the current TDD cycle"""
        if self.current_cycle:
            now = datetime.now()
            self.current_cycle['phases']['refactor_duration'] = (
                now - self.current_cycle['refactor_start']
            ).total_seconds()
            
            total_duration = sum(self.current_cycle['phases'].values())
            self.current_cycle['total_duration'] = total_duration
            
            self.cycles.append(self.current_cycle)
            self.current_cycle = None
    
    def get_cycle_statistics(self):
        """Get statistics about TDD cycles"""
        if not self.cycles:
            return {}
        
        red_times = [cycle['phases']['red_duration'] for cycle in self.cycles]
        green_times = [cycle['phases']['green_duration'] for cycle in self.cycles]
        refactor_times = [cycle['phases']['refactor_duration'] for cycle in self.cycles]
        total_times = [cycle['total_duration'] for cycle in self.cycles]
        
        return {
            'total_cycles': len(self.cycles),
            'average_red_time': sum(red_times) / len(red_times),
            'average_green_time': sum(green_times) / len(green_times),
            'average_refactor_time': sum(refactor_times) / len(refactor_times),
            'average_total_time': sum(total_times) / len(total_times),
            'fastest_cycle': min(total_times),
            'slowest_cycle': max(total_times)
        }

# Usage Example
tracker = TDDCycleTracker()

# Simulate TDD cycle
tracker.start_red_phase("Test user login with valid credentials")
time.sleep(2)  # Time spent writing failing test

tracker.start_green_phase()
time.sleep(5)  # Time spent making test pass

tracker.start_refactor_phase()
time.sleep(3)  # Time spent refactoring

tracker.complete_cycle()

stats = tracker.get_cycle_statistics()
print(f"Average cycle time: {stats['average_total_time']:.2f} seconds")
```

import unittest

class TestStringUtils(unittest.TestCase):

    def test_reverse_twice_returns_original(self, text):
        """Property: reversing a string twice should return the original"""

### Mutation Testing
```python
# mutation_tester.py
import ast
import copy
import subprocess
import tempfile
import os

class MutationTester:
    def __init__(self, source_file, test_file):
        self.source_file = source_file
        self.test_file = test_file
        self.mutations = []
    
    def generate_mutations(self):
        """Generate mutations of the source code"""
        with open(self.source_file, 'r') as f:
            source_code = f.read()
        
        tree = ast.parse(source_code)
        mutator = CodeMutator()
        
        # Generate different types of mutations
        mutations = []
        
        # Arithmetic operator mutations
        for node in ast.walk(tree):
            if isinstance(node, ast.Add):
                mutated_tree = copy.deepcopy(tree)
                # Replace Add with Sub
                self._replace_node(mutated_tree, node, ast.Sub())
                mutations.append(('arithmetic', ast.unparse(mutated_tree)))
        
        return mutations
    
    def run_mutation_tests(self):
        """Run tests against each mutation"""
        results = []
        
        for mutation_type, mutated_code in self.generate_mutations():
            # Write mutated code to temporary file
            with tempfile.NamedTemporaryFile(mode='w', suffix='.py', delete=False) as f:
                f.write(mutated_code)
                temp_file = f.name
            
            try:
                # Run tests against mutated code
                result = subprocess.run(
                    ['python', '-m', 'pytest', self.test_file],
                    capture_output=True,
                    text=True,
                    env={**os.environ, 'PYTHONPATH': os.path.dirname(temp_file)}
                )
                
                # If tests pass with mutation, the mutation survived (bad)
                # If tests fail with mutation, the mutation was killed (good)
                survived = result.returncode == 0
                
                results.append({
                    'mutation_type': mutation_type,
                    'survived': survived,
                    'output': result.stdout
                })
                
            finally:
                os.unlink(temp_file)
        
        return results
    
    def calculate_mutation_score(self, results):
        """Calculate mutation testing score"""
        total_mutations = len(results)
        killed_mutations = sum(1 for r in results if not r['survived'])
        
        if total_mutations == 0:
            return 0
        
        return (killed_mutations / total_mutations) * 100
```

```python
class TDDSuccessMetrics:
    def __init__(self):
        self.metrics = {}
    
    def track_test_metrics(self, project_data):
        """Track various TDD success metrics"""
        return {
            'test_coverage': self._calculate_coverage(project_data),
            'test_to_code_ratio': self._calculate_test_ratio(project_data),
            'defect_density': self._calculate_defect_density(project_data),
            'cycle_time': self._calculate_cycle_time(project_data),
            'team_confidence': self._measure_team_confidence(project_data)
        }
    
    def _calculate_coverage(self, data):
        return (data['lines_covered'] / data['total_lines']) * 100
    
    def _calculate_test_ratio(self, data):
        return data['test_lines'] / data['production_lines']
    
    def _calculate_defect_density(self, data):
        return data['defects_found'] / data['kloc']
    
    def _calculate_cycle_time(self, data):
        return data['total_development_time'] / data['features_delivered']
    
    def _measure_team_confidence(self, data):
        # Survey-based metric (1-10 scale)
        return data['confidence_survey_average']
```
