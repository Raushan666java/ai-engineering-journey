# Creational Design Patterns - Complete Implementation Guide
## Building Objects the Right Way

## 📋 Overview
Creational design patterns deal with object creation mechanisms, trying to create objects in a manner suitable to the situation. These patterns provide various object creation mechanisms, which increase flexibility and reuse of existing code.

## 🏭 Singleton Pattern

### Implementation and Use Cases
```python
import threading
from typing import Optional

class DatabaseConnection:
    """Thread-safe Singleton implementation"""
    _instance: Optional['DatabaseConnection'] = None
    _lock = threading.Lock()
    
    def __new__(cls) -> 'DatabaseConnection':
        if cls._instance is None:
            with cls._lock:
                if cls._instance is None:
                    cls._instance = super().__new__(cls)
                    cls._instance._initialized = False
        return cls._instance
    
    def __init__(self):
        if not self._initialized:
            self.connection_string = "postgresql://localhost:5432/mydb"
            self.connection_pool = []
            self.max_connections = 10
            self._initialized = True
    
    def get_connection(self):
        """Get database connection from pool"""
        if len(self.connection_pool) < self.max_connections:
            # Simulate creating new connection
            connection = f"Connection_{len(self.connection_pool) + 1}"
            self.connection_pool.append(connection)
            return connection
        else:
            # Return existing connection (round-robin)
            return self.connection_pool[len(self.connection_pool) % self.max_connections]
    
    def close_all_connections(self):
        """Close all database connections"""
        self.connection_pool.clear()

# Usage Example
db1 = DatabaseConnection()
db2 = DatabaseConnection()
print(db1 is db2)  # True - Same instance

# Thread-safe usage
import concurrent.futures

def get_db_connection():
    db = DatabaseConnection()
    return db.get_connection()

with concurrent.futures.ThreadPoolExecutor(max_workers=5) as executor:
    futures = [executor.submit(get_db_connection) for _ in range(10)]
    connections = [future.result() for future in futures]
```

```python
# Metaclass-based Singleton
class SingletonMeta(type):
    _instances = {}
    _lock = threading.Lock()
    
    def __call__(cls, *args, **kwargs):
        if cls not in cls._instances:
            with cls._lock:
                if cls not in cls._instances:
                    instance = super().__call__(*args, **kwargs)
                    cls._instances[cls] = instance
        return cls._instances[cls]

class Logger(metaclass=SingletonMeta):
    def __init__(self):
        self.log_level = "INFO"
        self.log_file = "application.log"
    
    def log(self, message, level="INFO"):
        print(f"[{level}] {message}")

# Usage
logger1 = Logger()
logger2 = Logger()
print(logger1 is logger2)  # True
```

## 🏗️ Factory Pattern

### Simple Factory
```python
from abc import ABC, abstractmethod
from typing import Dict, Type

class Vehicle(ABC):
    """Abstract base class for vehicles"""
    
    @abstractmethod
    def start(self) -> str:
        pass
    
    @abstractmethod
    def stop(self) -> str:
        pass
    
    @abstractmethod
    def get_info(self) -> Dict[str, str]:
        pass

class Car(Vehicle):
    def __init__(self, make: str, model: str):
        self.make = make
        self.model = model
        self.engine_running = False
    
    def start(self) -> str:
        self.engine_running = True
        return f"{self.make} {self.model} engine started"
    
    def stop(self) -> str:
        self.engine_running = False
        return f"{self.make} {self.model} engine stopped"
    
    def get_info(self) -> Dict[str, str]:
        return {
            'type': 'Car',
            'make': self.make,
            'model': self.model,
            'status': 'Running' if self.engine_running else 'Stopped'
        }

class Motorcycle(Vehicle):
    def __init__(self, make: str, model: str):
        self.make = make
        self.model = model
        self.engine_running = False
    
    def start(self) -> str:
        self.engine_running = True
        return f"{self.make} {self.model} motorcycle started"
    
    def stop(self) -> str:
        self.engine_running = False
        return f"{self.make} {self.model} motorcycle stopped"
    
    def get_info(self) -> Dict[str, str]:
        return {
            'type': 'Motorcycle',
            'make': self.make,
            'model': self.model,
            'status': 'Running' if self.engine_running else 'Stopped'
        }

class Truck(Vehicle):
    def __init__(self, make: str, model: str, payload_capacity: int):
        self.make = make
        self.model = model
        self.payload_capacity = payload_capacity
        self.engine_running = False
    
    def start(self) -> str:
        self.engine_running = True
        return f"{self.make} {self.model} truck started"
    
    def stop(self) -> str:
        self.engine_running = False
        return f"{self.make} {self.model} truck stopped"
    
    def get_info(self) -> Dict[str, str]:
        return {
            'type': 'Truck',
            'make': self.make,
            'model': self.model,
            'payload_capacity': f"{self.payload_capacity} tons",
            'status': 'Running' if self.engine_running else 'Stopped'
        }

class VehicleFactory:
    """Simple Factory for creating vehicles"""
    
    _vehicle_types: Dict[str, Type[Vehicle]] = {
        'car': Car,
        'motorcycle': Motorcycle,
        'truck': Truck
    }
    
    @classmethod
    def create_vehicle(cls, vehicle_type: str, **kwargs) -> Vehicle:
        """Create vehicle based on type"""
        vehicle_class = cls._vehicle_types.get(vehicle_type.lower())
        
        if not vehicle_class:
            raise ValueError(f"Unknown vehicle type: {vehicle_type}")
        
        return vehicle_class(**kwargs)
    
    @classmethod
    def register_vehicle_type(cls, vehicle_type: str, vehicle_class: Type[Vehicle]):
        """Register new vehicle type"""
        cls._vehicle_types[vehicle_type.lower()] = vehicle_class
    
    @classmethod
    def get_available_types(cls) -> list:
        """Get list of available vehicle types"""
        return list(cls._vehicle_types.keys())

# Usage Example
factory = VehicleFactory()

# Create different vehicles
car = factory.create_vehicle('car', make='Toyota', model='Camry')
motorcycle = factory.create_vehicle('motorcycle', make='Honda', model='CBR600RR')
truck = factory.create_vehicle('truck', make='Ford', model='F-150', payload_capacity=2)

print(car.start())
print(motorcycle.start())
print(truck.get_info())
```

### Factory Method Pattern
```python
class VehicleManufacturer(ABC):
    """Abstract manufacturer class"""
    
    @abstractmethod
    def create_vehicle(self, model: str) -> Vehicle:
        """Factory method to create vehicle"""
        pass
    
    def manufacture_vehicle(self, model: str) -> Vehicle:
        """Template method using factory method"""
        vehicle = self.create_vehicle(model)
        self.quality_check(vehicle)
        self.register_vehicle(vehicle)
        return vehicle
    
    def quality_check(self, vehicle: Vehicle):
        """Perform quality check"""
        print(f"Quality check passed for {vehicle.get_info()['type']}")
    
    def register_vehicle(self, vehicle: Vehicle):
        """Register vehicle in system"""
        print(f"Vehicle registered: {vehicle.get_info()}")

class ToyotaManufacturer(VehicleManufacturer):
    """Concrete Toyota manufacturer"""
    
    def create_vehicle(self, model: str) -> Vehicle:
        return Car(make='Toyota', model=model)

class HondaManufacturer(VehicleManufacturer):
    """Concrete Honda manufacturer"""
    
    def create_vehicle(self, model: str) -> Vehicle:
        if model.startswith('CBR'):
            return Motorcycle(make='Honda', model=model)
        else:
            return Car(make='Honda', model=model)

class FordManufacturer(VehicleManufacturer):
    """Concrete Ford manufacturer"""
    
    def create_vehicle(self, model: str) -> Vehicle:
        if 'F-' in model:
            return Truck(make='Ford', model=model, payload_capacity=3)
        else:
            return Car(make='Ford', model=model)

# Usage
toyota_factory = ToyotaManufacturer()
honda_factory = HondaManufacturer()
ford_factory = FordManufacturer()

camry = toyota_factory.manufacture_vehicle('Camry')
cbr = honda_factory.manufacture_vehicle('CBR600RR')
f150 = ford_factory.manufacture_vehicle('F-150')
```

### Abstract Factory Pattern
```python
class GUIFactory(ABC):
    """Abstract factory for GUI components"""
    
    @abstractmethod
    def create_button(self) -> 'Button':
        pass
    
    @abstractmethod
    def create_checkbox(self) -> 'Checkbox':
        pass
    
    @abstractmethod
    def create_textfield(self) -> 'TextField':
        pass

class Button(ABC):
    @abstractmethod
    def render(self) -> str:
        pass
    
    @abstractmethod
    def click(self) -> str:
        pass

class Checkbox(ABC):
    @abstractmethod
    def render(self) -> str:
        pass
    
    @abstractmethod
    def toggle(self) -> str:
        pass

class TextField(ABC):
    @abstractmethod
    def render(self) -> str:
        pass
    
    @abstractmethod
    def set_text(self, text: str) -> str:
        pass

# Windows implementations
class WindowsButton(Button):
    def render(self) -> str:
        return "Rendering Windows-style button"
    
    def click(self) -> str:
        return "Windows button clicked"

class WindowsCheckbox(Checkbox):
    def __init__(self):
        self.checked = False
    
    def render(self) -> str:
        return "Rendering Windows-style checkbox"
    
    def toggle(self) -> str:
        self.checked = not self.checked
        return f"Windows checkbox {'checked' if self.checked else 'unchecked'}"

class WindowsTextField(TextField):
    def __init__(self):
        self.text = ""
    
    def render(self) -> str:
        return "Rendering Windows-style text field"
    
    def set_text(self, text: str) -> str:
        self.text = text
        return f"Windows text field set to: {text}"

# macOS implementations
class MacButton(Button):
    def render(self) -> str:
        return "Rendering macOS-style button"
    
    def click(self) -> str:
        return "macOS button clicked"

class MacCheckbox(Checkbox):
    def __init__(self):
        self.checked = False
    
    def render(self) -> str:
        return "Rendering macOS-style checkbox"
    
    def toggle(self) -> str:
        self.checked = not self.checked
        return f"macOS checkbox {'checked' if self.checked else 'unchecked'}"

class MacTextField(TextField):
    def __init__(self):
        self.text = ""
    
    def render(self) -> str:
        return "Rendering macOS-style text field"
    
    def set_text(self, text: str) -> str:
        self.text = text
        return f"macOS text field set to: {text}"

# Concrete factories
class WindowsGUIFactory(GUIFactory):
    def create_button(self) -> Button:
        return WindowsButton()
    
    def create_checkbox(self) -> Checkbox:
        return WindowsCheckbox()
    
    def create_textfield(self) -> TextField:
        return WindowsTextField()

class MacGUIFactory(GUIFactory):
    def create_button(self) -> Button:
        return MacButton()
    
    def create_checkbox(self) -> Checkbox:
        return MacCheckbox()
    
    def create_textfield(self) -> TextField:
        return MacTextField()

# Client code
class Application:
    def __init__(self, factory: GUIFactory):
        self.factory = factory
        self.button = factory.create_button()
        self.checkbox = factory.create_checkbox()
        self.textfield = factory.create_textfield()
    
    def render_ui(self):
        return [
            self.button.render(),
            self.checkbox.render(),
            self.textfield.render()
        ]

# Usage
def create_application(os_type: str) -> Application:
    if os_type.lower() == 'windows':
        return Application(WindowsGUIFactory())
    elif os_type.lower() == 'macos':
        return Application(MacGUIFactory())
    else:
        raise ValueError(f"Unsupported OS: {os_type}")

# Create applications for different OS

mac_app = create_application('macos')

print("Windows UI:")
for component in windows_app.render_ui():
    print(f"  {component}")

print("\nmacOS UI:")
for component in mac_app.render_ui():
    print(f"  {component}")
```

```python
from typing import Optional, List

class Computer:
    """Product class"""
    
    def __init__(self):
        self.cpu: Optional[str] = None
        self.memory: Optional[str] = None
        self.storage: Optional[str] = None
        self.graphics: Optional[str] = None
        self.motherboard: Optional[str] = None
        self.power_supply: Optional[str] = None
        self.case: Optional[str] = None
        self.cooling: Optional[str] = None
        self.network: Optional[str] = None
        self.accessories: List[str] = []
    
    def __str__(self) -> str:
        specs = []
        if self.cpu: specs.append(f"CPU: {self.cpu}")
        if self.memory: specs.append(f"Memory: {self.memory}")
        if self.storage: specs.append(f"Storage: {self.storage}")
        if self.graphics: specs.append(f"Graphics: {self.graphics}")
        if self.motherboard: specs.append(f"Motherboard: {self.motherboard}")
        if self.power_supply: specs.append(f"Power Supply: {self.power_supply}")
        if self.case: specs.append(f"Case: {self.case}")
        if self.cooling: specs.append(f"Cooling: {self.cooling}")
        if self.network: specs.append(f"Network: {self.network}")
        if self.accessories: specs.append(f"Accessories: {', '.join(self.accessories)}")
        
        return "Computer Specifications:\n" + "\n".join(f"  {spec}" for spec in specs)

class ComputerBuilder:
    """Builder class with fluent interface"""
    
    def __init__(self):
        self.computer = Computer()
    
    def set_cpu(self, cpu: str) -> 'ComputerBuilder':
        self.computer.cpu = cpu
        return self
    
    def set_memory(self, memory: str) -> 'ComputerBuilder':
        self.computer.memory = memory
        return self
    
    def set_storage(self, storage: str) -> 'ComputerBuilder':
        self.computer.storage = storage

        self.computer.graphics = graphics
        return self
    
    def set_motherboard(self, motherboard: str) -> 'ComputerBuilder':
        self.computer.motherboard = motherboard
        return self
    
    def set_power_supply(self, power_supply: str) -> 'ComputerBuilder':
        self.computer.power_supply = power_supply
        return self
    
    def set_case(self, case: str) -> 'ComputerBuilder':
        self.computer.case = case
        return self

        self.computer.cooling = cooling
        return self

        self.computer.network = network
        return self
    
    def add_accessory(self, accessory: str) -> 'ComputerBuilder':
        self.computer.accessories.append(accessory)

        # Validation
        if not self.computer.cpu:
            raise ValueError("CPU is required")
        if not self.computer.memory:
            raise ValueError("Memory is required")
        if not self.computer.storage:
            raise ValueError("Storage is required")
        
        return self.computer
```

## 🔄 Prototype Pattern

### Basic Prototype Implementation
```python
import copy
from abc import ABC, abstractmethod
from typing import Dict, Any

class Prototype(ABC):
    """Abstract prototype interface"""
    
    @abstractmethod
    def clone(self) -> 'Prototype':
        pass

class Document(Prototype):
    """Concrete prototype for documents"""
    
    def __init__(self, title: str, content: str, metadata: Dict[str, Any] = None):
        self.title = title
        self.content = content
        self.metadata = metadata or {}
        self.creation_date = "2024-01-01"
        self.version = 1.0
        self.formatting = {
            'font': 'Arial',
            'size': 12,
            'color': 'black'
        }
    
    def clone(self) -> 'Document':
        """Create a deep copy of the document"""
        # Deep copy to ensure nested objects are also copied
        cloned = copy.deepcopy(self)
        
        # Modify some properties for the clone
        cloned.title = f"Copy of {self.title}"
        cloned.version = self.version + 0.1
        
        return cloned
    
    def set_title(self, title: str):
        self.title = title
    
    def set_content(self, content: str):
        self.content = content
    
    def add_metadata(self, key: str, value: Any):
        self.metadata[key] = value
    
    def set_formatting(self, **kwargs):
        self.formatting.update(kwargs)
    
    def __str__(self) -> str:
        return f"""Document: {self.title}
Version: {self.version}
Content: {self.content[:50]}{'...' if len(self.content) > 50 else ''}
Metadata: {self.metadata}
Formatting: {self.formatting}"""

class Presentation(Prototype):
    """Concrete prototype for presentations"""
    
    def __init__(self, title: str, slides: List[str] = None, theme: str = "default"):
        self.title = title
        self.slides = slides or []
        self.theme = theme
        self.template_settings = {
            'background_color': 'white',
            'font_family': 'Calibri',
            'slide_transition': 'fade'
        }
        self.speaker_notes = {}
    
    def clone(self) -> 'Presentation':
        """Create a deep copy of the presentation"""
        cloned = copy.deepcopy(self)
        cloned.title = f"Copy of {self.title}"
        return cloned
    
    def add_slide(self, slide_content: str):
        self.slides.append(slide_content)
    
    def set_theme(self, theme: str):
        self.theme = theme
    
    def add_speaker_note(self, slide_index: int, note: str):
        self.speaker_notes[slide_index] = note
    
    def __str__(self) -> str:
        return f"""Presentation: {self.title}
Theme: {self.theme}
Slides: {len(self.slides)}
Template Settings: {self.template_settings}"""

class PrototypeRegistry:
    """Registry to manage prototypes"""
    
    def __init__(self):
        self._prototypes: Dict[str, Prototype] = {}
    
    def register_prototype(self, name: str, prototype: Prototype):
        """Register a prototype with a name"""
        self._prototypes[name] = prototype
    
    def unregister_prototype(self, name: str):
        """Remove a prototype from registry"""
        if name in self._prototypes:
            del self._prototypes[name]
    
    def clone_prototype(self, name: str) -> Prototype:
        """Clone a registered prototype"""
        if name not in self._prototypes:
            raise ValueError(f"Prototype '{name}' not found in registry")
        
        return self._prototypes[name].clone()
    
    def list_prototypes(self) -> List[str]:
        """List all registered prototype names"""
        return list(self._prototypes.keys())

# Usage Example
registry = PrototypeRegistry()

# Create and register prototypes
business_doc = Document(
    title="Business Report Template",
    content="Executive Summary\n\n1. Introduction\n2. Analysis\n3. Recommendations\n4. Conclusion",
    metadata={"department": "Business", "confidentiality": "Internal"}
)
business_doc.set_formatting(font='Times New Roman', size=11)

tech_presentation = Presentation(
    title="Technical Presentation Template",
    slides=["Title Slide", "Agenda", "Problem Statement", "Solution", "Implementation", "Q&A"],
    theme="corporate"
)
tech_presentation.template_settings.update({
    'background_color': 'navy',
    'font_family': 'Arial',
    'slide_transition': 'slide'
})

# Register prototypes
registry.register_prototype("business_report", business_doc)
registry.register_prototype("tech_presentation", tech_presentation)

# Clone prototypes to create new documents
quarterly_report = registry.clone_prototype("business_report")
quarterly_report.set_title("Q4 2024 Business Report")
quarterly_report.set_content("Q4 Executive Summary\n\n1. Q4 Performance\n2. Market Analysis\n3. Strategic Recommendations\n4. 2025 Outlook")
quarterly_report.add_metadata("quarter", "Q4 2024")

project_presentation = registry.clone_prototype("tech_presentation")
project_presentation.set_title("Project Alpha Technical Review")
project_presentation.add_slide("Technical Architecture")
project_presentation.add_slide("Performance Metrics")
project_presentation.add_speaker_note(0, "Welcome everyone to the technical review")

print("Original Business Document:")
print(business_doc)
print("\nCloned Quarterly Report:")
print(quarterly_report)
print("\nOriginal Tech Presentation:")
print(tech_presentation)
print("\nCloned Project Presentation:")
print(project_presentation)
```

### Advanced Prototype with Caching
```python
import time
from typing import Optional
import hashlib

class CachedPrototype(Prototype):
    """Prototype with caching capabilities"""
    
    def __init__(self):
        self._cache: Dict[str, 'CachedPrototype'] = {}
        self._cache_ttl = 300  # 5 minutes
        self._cache_timestamps: Dict[str, float] = {}
    
    def clone(self, cache_key: Optional[str] = None) -> 'CachedPrototype':
        """Clone with optional caching"""
        if cache_key:
            # Check if cached version exists and is still valid
            if (cache_key in self._cache and 
                cache_key in self._cache_timestamps and
                time.time() - self._cache_timestamps[cache_key] < self._cache_ttl):
                return copy.deepcopy(self._cache[cache_key])
        
        # Create new clone
        cloned = copy.deepcopy(self)
        
        # Cache the clone if cache_key provided
        if cache_key:
            self._cache[cache_key] = copy.deepcopy(cloned)
            self._cache_timestamps[cache_key] = time.time()
        
        return cloned
    
    def clear_cache(self):
        """Clear all cached prototypes"""
        self._cache.clear()
        self._cache_timestamps.clear()
    
    def get_cache_info(self) -> Dict[str, Any]:
        """Get cache statistics"""
        current_time = time.time()
        valid_entries = sum(1 for timestamp in self._cache_timestamps.values() 
                          if current_time - timestamp < self._cache_ttl)
        
        return {
            'total_entries': len(self._cache),
            'valid_entries': valid_entries,
            'expired_entries': len(self._cache) - valid_entries,
            'cache_ttl': self._cache_ttl
        }

class OptimizedDocument(Document, CachedPrototype):
    """Document with caching and optimization"""
    
    def __init__(self, title: str, content: str, metadata: Dict[str, Any] = None):
        Document.__init__(self, title, content, metadata)
        CachedPrototype.__init__(self)
        self._content_hash = self._calculate_hash()
    
    def _calculate_hash(self) -> str:
        """Calculate hash of document content for change detection"""
        content_str = f"{self.title}{self.content}{str(self.metadata)}"
        return hashlib.md5(content_str.encode()).hexdigest()
    
    def set_content(self, content: str):
        """Override to update hash when content changes"""
        super().set_content(content)
        self._content_hash = self._calculate_hash()
        # Clear cache when content changes
        self.clear_cache()
    
    def clone(self, cache_key: Optional[str] = None) -> 'OptimizedDocument':
        """Optimized cloning with hash-based caching"""
        if not cache_key:
            cache_key = self._content_hash
        
        return super().clone(cache_key)

# Usage Example with Caching
optimized_doc = OptimizedDocument(
    title="Cached Document Template",
    content="This is a template document that will be cloned frequently.",
    metadata={"type": "template", "version": "1.0"}
)

# Clone multiple times - first clone creates cache entry

# Subsequent clones use cached version

print(f"First clone time: {first_clone_time:.6f} seconds")
print(f"Cached clone time: {cached_clone_time:.6f} seconds")
print(f"Cache info: {optimized_doc.get_cache_info()}")
```

## 🎯 Creational Patterns Best Practices

### Pattern Selection Guide
```python
class CreationalPatternSelector:
    """Helper class to select appropriate creational pattern"""
    
    def __init__(self):
        self.selection_criteria = {
            'singleton': {
                'use_when': [
                    'Need exactly one instance',
                    'Global access point required',
                    'Expensive object creation',
                    'Shared resource management'
                ],
                'avoid_when': [
                    'Multiple instances might be needed',
                    'Testing is difficult',
                    'Tight coupling concerns'
                ]
            },
            'factory_method': {
                'use_when': [
                    'Object creation logic is complex',
                    'Need to delegate creation to subclasses',
                    'Product types determined at runtime',
                    'Want to localize object creation'
                ],
                'avoid_when': [
                    'Simple object creation',
                    'No variation in products',
                    'Unnecessary abstraction'
                ]
            },
            'abstract_factory': {
                'use_when': [
                    'Need families of related objects',
                    'Multiple product lines',
                    'Platform-specific implementations',
                    'Consistency across products required'
                ],
                'avoid_when': [
                    'Single product type',
                    'No product families',
                    'Simple object creation'
                ]
            },
            'builder': {
                'use_when': [
                    'Complex object construction',
                    'Many optional parameters',
                    'Step-by-step construction needed',
                    'Different representations of same object'
                ],
                'avoid_when': [
                    'Simple object creation',
                    'Few parameters',
                    'No construction variations'
                ]
            },
            'prototype': {
                'use_when': [
                    'Expensive object creation',
                    'Similar objects with slight variations',
                    'Runtime object composition',
                    'Avoiding subclass explosion'
                ],
                'avoid_when': [
                    'Simple object creation',
                    'Deep copying is expensive',
                    'Objects don\'t vary much'
                ]
            }
        }
    
    def recommend_pattern(self, requirements: List[str]) -> List[str]:
        """Recommend patterns based on requirements"""
        recommendations = []
        
        for pattern, criteria in self.selection_criteria.items():
            score = 0
            for req in requirements:
                if any(req.lower() in use_case.lower() for use_case in criteria['use_when']):
                    score += 1
                if any(req.lower() in avoid_case.lower() for avoid_case in criteria['avoid_when']):
                    score -= 1
            
            if score > 0:
                recommendations.append((pattern, score))
        
        # Sort by score and return pattern names
        recommendations.sort(key=lambda x: x[1], reverse=True)
        return [pattern for pattern, score in recommendations]

# Usage
selector = CreationalPatternSelector()
requirements = [
    "complex object construction",
    "many optional parameters",
    "step-by-step construction"
]

recommended_patterns = selector.recommend_pattern(requirements)
print(f"Recommended patterns: {recommended_patterns}")
```

---

*Creational patterns provide the foundation for flexible object creation. Choose the right pattern based on your specific needs: Singleton for single instances, Factory for creation logic, Builder for complex construction, and Prototype for copying expensive objects.*
