TypeScript Implementations​

The \WeakTopicAnalyzer\ class identifies a student's weakest GATE subjects from mock test performance data.

\\ ypescript /**

  • WeakTopicAnalyzer — Analyzes mock test results
  • to identify weakest subjects and suggest focus areas. */ interface TopicScore { topic: string; correct: number; wrong: number; total: number; }

class WeakTopicAnalyzer { private topics: Map<string, { correct: number; wrong: number; total: number }> = new Map();

/** Record performance for a topic. */ recordTopic(topic: string, correct: number, wrong: number, total: number): void { const existing = this.topics.get(topic) ?? { correct: 0, wrong: 0, total: 0 }; existing.correct += correct; existing.wrong += wrong; existing.total += total; this.topics.set(topic, existing); }

/** Accuracy per topic (%). */ topicAccuracy(topic: string): number { const data = this.topics.get(topic); if (!data || data.total === 0) return 0; return Math.round((data.correct / data.total) * 10000) / 100; }

/** Net score per topic (with negative marking). */ topicNetScore(topic: string, marksPerQ: number = 1, penalty: number = 0.25): number { const data = this.topics.get(topic); if (!data) return 0; return data.correct * marksPerQ - data.wrong * penalty; }

/** Rank topics by weakness (ascending accuracy). */ weakestTopics(topN: number = 3): { topic: string; accuracy: number; netScore: number }[] { const entries: { topic: string; accuracy: number; netScore: number }[] = []; for (const [topic] of this.topics) { entries.push({ topic, accuracy: this.topicAccuracy(topic), netScore: this.topicNetScore(topic), }); } entries.sort((a, b) => a.accuracy - b.accuracy); return entries.slice(0, topN); }

/** Generate an improvement plan based on weak areas. */ improvementPlan(): string[] { const weak = this.weakestTopics(3); const recommendations: string[] = []; for (const w of weak) { if (w.accuracy < 40) { recommendations.push(\: Accuracy is critically low (%). Revise fundamentals and practice 20+ problems.); } else if (w.accuracy < 60) { recommendations.push(\: Below average (%). Focus on past GATE questions and concept clarity.); } else if (w.accuracy < 75) { recommendations.push(\: Moderate (%). Improve speed with timed practice.); } else { recommendations.push(\: Good (%). Maintain with weekly review.); } } return recommendations; }

/** Overall accuracy across all topics. */ overallAccuracy(): number { let correct = 0, total = 0; for (const [, data] of this.topics) { correct += data.correct; total += data.total; } return total > 0 ? Math.round((correct / total) * 10000) / 100 : 0; } }
