class ProductBacklog:

            'id': len(self.backlog_items) + 1,
            'user_story': user_story,
            'priority': priority,
            'business_value': business_value,
            'story_points': None,
            'acceptance_criteria': [],
            'definition_of_ready_met': False,
            'dependencies': [],
            'created_date': datetime.now()
        }
        
        self.backlog_items.append(backlog_item)
        return backlog_item
    
    def refine_backlog(self):
        """Ongoing backlog refinement activities"""
        refinement_activities = {
            'story_breakdown': self._break_down_large_stories(),
            'acceptance_criteria': self._add_acceptance_criteria(),
            'estimation': self._estimate_story_points(),
            'prioritization': self._reprioritize_items(),
            'dependency_analysis': self._analyze_dependencies()
        }
        
        return refinement_activities
    
    def prioritize_backlog(self, prioritization_method='business_value'):
        """Prioritize product backlog items"""
        if prioritization_method == 'business_value':
            self.backlog_items.sort(key=lambda x: x['business_value'], reverse=True)
        elif prioritization_method == 'moscow':
            priority_order = {'must': 1, 'should': 2, 'could': 3, 'wont': 4}
            self.backlog_items.sort(key=lambda x: priority_order.get(x['priority'], 5))
        
        return self.backlog_items
```

class VelocityTracker:
    def __init__(self):
        self.sprint_velocities = []
        self.team_capacity_history = []
    
    def record_sprint_velocity(self, sprint_number, story_points_completed, team_capacity):
        """Record velocity for completed sprint"""
        velocity_record = {
            'sprint_number': sprint_number,
            'story_points_completed': story_points_completed,
            'team_capacity': team_capacity,
            'velocity_per_capacity': story_points_completed / team_capacity if team_capacity > 0 else 0,
            'date': datetime.now()
        }
        
        self.sprint_velocities.append(velocity_record)
        return velocity_record
    
    def calculate_average_velocity(self, last_n_sprints=3):
        """Calculate average velocity over last N sprints"""
        recent_sprints = self.sprint_velocities[-last_n_sprints:]
        if not recent_sprints:
            return 0
        
        total_points = sum(sprint['story_points_completed'] for sprint in recent_sprints)
        return total_points / len(recent_sprints)
    
    def predict_completion(self, remaining_story_points):
        """Predict completion based on average velocity"""
        avg_velocity = self.calculate_average_velocity()
        if avg_velocity == 0:
            return None
        
        sprints_needed = remaining_story_points / avg_velocity
        return {
            'sprints_needed': sprints_needed,
            'estimated_completion': self._calculate_completion_date(sprints_needed)
        }
```

### Burndown Charts
```python
class BurndownChart:
    def __init__(self, total_story_points, sprint_length_days):
        self.total_story_points = total_story_points
        self.sprint_length = sprint_length_days
        self.daily_remaining = [total_story_points]
        self.ideal_burndown = self._calculate_ideal_burndown()
    
    def _calculate_ideal_burndown(self):
        """Calculate ideal burndown line"""
        daily_burn_rate = self.total_story_points / self.sprint_length
        ideal_line = []
        
        for day in range(self.sprint_length + 1):
            remaining = max(0, self.total_story_points - (day * daily_burn_rate))
            ideal_line.append(remaining)
        
        return ideal_line
    
    def update_remaining_work(self, story_points_completed):
        """Update remaining work for the day"""
        current_remaining = self.daily_remaining[-1]
        new_remaining = max(0, current_remaining - story_points_completed)
        self.daily_remaining.append(new_remaining)
    
    def analyze_burndown_trend(self):
        """Analyze burndown trend and provide insights"""
        if len(self.daily_remaining) < 3:
            return "Insufficient data for trend analysis"
        
        current_day = len(self.daily_remaining) - 1
        actual_remaining = self.daily_remaining[-1]
        ideal_remaining = self.ideal_burndown[current_day]
        
        if actual_remaining > ideal_remaining:
            return "Behind schedule - consider scope adjustment or team support"
        elif actual_remaining < ideal_remaining:
            return "Ahead of schedule - consider adding scope or helping other teams"
        else:
            return "On track with ideal burndown"
```
