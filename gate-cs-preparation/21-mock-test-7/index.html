  • question bank, enforces time limits, and scores responses. */ interface Question { id: number; text: string; subject: string; difficulty: 1 | 2 | 3; marks: number; negativeMark: number; correctAnswer: string; }

interface SimResult { answered: number; correct: number; wrong: number; rawScore: number; timeTaken: number; accuracy: number; }

class MockTestSimulator { private questions: Question[] = []; private selected: Question[] = []; private answers: Map<number, string> = new Map(); private startTime: number = 0; private isRunning: boolean = false;

/** Load question bank. */ loadBank(questions: Question[]): void { this.questions = questions; }

/** Select N random questions, optionally filtered by subject. */ selectRandom(count: number, subject?: string): Question[] { let pool = this.questions; if (subject) pool = pool.filter((q) => q.subject === subject); this.selected = []; const shuffled = [...pool].sort(() => Math.random() - 0.5); for (let i = 0; i < Math.min(count, shuffled.length); i++) { this.selected.push(shuffled[i]); } return this.selected; }

/** Start the mock test timer. */ start(): void { this.startTime = Date.now(); this.isRunning = true; this.answers.clear(); }

/** Submit an answer for a question. */ answerQuestion(qId: number, answer: string): void { if (!this.isRunning) throw new Error('Test not started'); this.answers.set(qId, answer); }

/** End the test and compute results. */ endTest(): SimResult { if (!this.isRunning) throw new Error('Test not started'); const timeTaken = Math.round((Date.now() - this.startTime) / 1000); this.isRunning = false;

let correct = 0, wrong = 0;
for (const q of this.selected) {
  const ans = this.answers.get(q.id);
  if (!ans) continue; // unattempted
  if (ans === q.correctAnswer) correct++;
  else wrong++;
}

let rawScore = 0;
for (const q of this.selected) {
  const ans = this.answers.get(q.id);
  if (ans === q.correctAnswer) rawScore += q.marks;
  else if (ans !== undefined) rawScore -= q.negativeMark;
}

return {
  answered: this.answers.size,
  correct,
  wrong,
  rawScore: Math.round(rawScore * 100) / 100,
  timeTaken,
  accuracy: this.answers.size > 0
    ? Math.round((correct / this.answers.size) * 10000) / 100
    : 0,
};

}
