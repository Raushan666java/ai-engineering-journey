# ai_teacher/teaching_engine.py

import openai
from typing import List, Dict
import json

class AiTeacherEngine:
    def __init__(self, api_key: str):
        self.api_key = api_key
        openai.api_key = api_key
        self.conversation_history = []
        
    def load_master_prompt(self, subject: str = None):
        """Load appropriate system prompt"""
        prompt = MASTER_SYSTEM_PROMPT
        if subject and subject in SUBJECT_SPECIFIC_PROMPTS:
            prompt += f"\n\n{SUBJECT_SPECIFIC_PROMPTS[subject]}"
        return prompt
    
    def assess_knowledge(self, student_id: int, topic: str):
        """Initial knowledge assessment"""
        assessment_prompt = f"""
        Assess the student's current knowledge of {topic}.
        Ask 3 progressively difficult questions to gauge their level.
        Classify as: Beginner, Intermediate, or Advanced.
        """
        return self.teach(assessment_prompt, subject=topic)
    
    def teach(self, user_input: str, subject: str = None):
        """Main teaching interaction"""
        system_prompt = self.load_master_prompt(subject)
        
        messages = [
            {"role": "system", "content": system_prompt},
            *self.conversation_history,
            {"role": "user", "content": user_input}
        ]
        
        response = openai.ChatCompletion.create(
            model="gpt-4",
            messages=messages,
            temperature=0.7,
            max_tokens=1000
        )
        
        assistant_message = response.choices[0].message.content
        
        # Store conversation
        self.conversation_history.append(
            {"role": "user", "content": user_input}
        )
        self.conversation_history.append(
            {"role": "assistant", "content": assistant_message}
        )
        
        return assistant_message
    
    def generate_practice_problems(self, topic: str, difficulty: str, count: int = 3):
        """Generate practice problems based on topic"""
        prompt = f"""
        Generate {count} {difficulty} level practice problems for {topic}.
        Include:
        1. Problem statement
        2. Example input/output
        3. Constraints
        4. Hints (don't reveal solution)
        """
        return self.teach(prompt, subject=topic)
    
    def provide_feedback(self, student_solution: str, problem: str):
        """Provide detailed feedback on student solution"""
        prompt = f"""
        Problem: {problem}
        Student Solution: {student_solution}
        
        Provide constructive feedback:
        1. What's done well
        2. Areas for improvement
        3. Optimization suggestions
        4. Time/space complexity analysis
        
        Be encouraging and educational.
        """
        return self.teach(prompt)

# Usage Example
if __name__ == "__main__":

        student_id=1,

    )
    print(response)

    next_response = teacher.teach(student_response, subject="dsa")
