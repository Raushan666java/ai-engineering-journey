```python
class MobProgramming:

                'activities': [
                    'Review what was accomplished',
                    'Discuss what worked well',
                    'Identify improvements for next session',
                    'Plan follow-up actions'
                ]
            }
        }
    
    def facilitate_mob_session(self, team_size, session_duration, task):
        """Facilitate effective mob programming session"""
        cycles = (session_duration - 25) // 15  # Account for setup and retro
        
        return {
            'team_size': team_size,
            'total_duration': session_duration,
            'work_cycles': cycles,
            'rotation_schedule': self._create_rotation_schedule(team_size, cycles),
            'facilitation_tips': [
                'Keep energy high with regular breaks',
                'Encourage all voices to be heard',
                'Focus on learning and knowledge sharing',
                'Document decisions and learnings'
            ],
            'tools_needed': [
                'Shared screen/projector',
                'Version control system',
                'Communication tools',
                'Timer for rotations'
            ]
        }
    
    def _create_rotation_schedule(self, team_size, cycles):
        """Create rotation schedule for mob session"""
        roles = ['driver', 'navigator'] + [f'mob_member_{i}' for i in range(team_size - 2)]
        schedule = []
        
        for cycle in range(cycles):
            cycle_roles = {}
            for i, role in enumerate(roles):
                person_index = (cycle + i) % team_size
                cycle_roles[f'person_{person_index}'] = role
            schedule.append(cycle_roles)
        
        return schedule
```

### Review Feedback Framework
```python
class ReviewFeedback:
    def __init__(self):
        self.feedback_types = {
            'must_fix': {
                'prefix': '🚨 MUST FIX:',
                'description': 'Critical issues that must be addressed',
                'examples': ['Security vulnerabilities', 'Functional bugs', 'Breaking changes']
            },
            'should_fix': {
                'prefix': '⚠️ SHOULD FIX:',
                'description': 'Important issues that should be addressed',
                'examples': ['Performance issues', 'Design problems', 'Maintainability concerns']
            },
            'consider': {
                'prefix': '💭 CONSIDER:',
                'description': 'Suggestions for improvement',
                'examples': ['Alternative approaches', 'Code style improvements', 'Optimization opportunities']
            },
            'praise': {
                'prefix': '👍 GOOD:',
                'description': 'Positive feedback on good practices',
                'examples': ['Clean code', 'Good test coverage', 'Clever solutions']
            },
            'question': {
                'prefix': '❓ QUESTION:',
                'description': 'Questions for clarification',
                'examples': ['Why this approach?', 'Is this the expected behavior?', 'Can you explain this logic?']
            }
        }
    
    def format_feedback(self, feedback_type, message, code_snippet=None):
        """Format feedback according to established conventions"""
        prefix = self.feedback_types[feedback_type]['prefix']
        formatted_feedback = f"{prefix} {message}"
        
        if code_snippet:
            formatted_feedback += f"\n\n```\n{code_snippet}\n```"
        
        return formatted_feedback
    
    def generate_review_summary(self, feedback_items):
        """Generate summary of review feedback"""
        summary = {
            'total_comments': len(feedback_items),
            'by_type': {},
            'action_required': False
        }
        
        for item in feedback_items:
            feedback_type = item['type']
            summary['by_type'][feedback_type] = summary['by_type'].get(feedback_type, 0) + 1
            
            if feedback_type in ['must_fix', 'should_fix']:
                summary['action_required'] = True
        
        return summary
```

```python
    def measure_team_health(self, team_data):
        """Measure overall team collaboration health"""
        scores = {}
        
        # Communication metrics
        scores['communication'] = self._calculate_communication_score(team_data)
        
        # Code collaboration metrics
        scores['code_collaboration'] = self._calculate_code_collaboration_score(team_data)
        
        # Team dynamics (survey-based)
        scores['team_dynamics'] = self._calculate_team_dynamics_score(team_data)
        
        # Delivery metrics
        scores['delivery'] = self._calculate_delivery_score(team_data)
        
        overall_score = sum(scores.values()) / len(scores)
        
        return {
            'overall_score': overall_score,
            'category_scores': scores,
            'recommendations': self._generate_recommendations(scores)
        }
    
    def _calculate_communication_score(self, data):
        """Calculate communication effectiveness score"""
        response_time_score = min(100, (24 - data.get('avg_response_hours', 24)) / 24 * 100)
        meeting_efficiency_score = data.get('meeting_efficiency_percent', 0)
        
        return (response_time_score + meeting_efficiency_score) / 2
    
    def _generate_recommendations(self, scores):
        """Generate improvement recommendations based on scores"""
        recommendations = []
        
        if scores['communication'] < 70:
            recommendations.append("Improve communication channels and response times")
        
        if scores['code_collaboration'] < 70:
            recommendations.append("Increase pair programming and code review participation")
        
        if scores['team_dynamics'] < 70:
            recommendations.append("Focus on building psychological safety and trust")
        
        if scores['delivery'] < 70:
            recommendations.append("Streamline delivery process and reduce dependencies")
        
        return recommendations
```

```python
            'action_items': self._generate_action_items(gaps),
            'timeline': self._create_timeline(gaps, constraints),
            'success_metrics': self._define_success_metrics(gaps),
            'resource_requirements': self._estimate_resources(gaps)
        }
    
    def _identify_gaps(self, current, desired):
        """Identify gaps between current and desired state"""
        gaps = []
        for area in desired:
            if area not in current or current[area] < desired[area]:
                gaps.append({
                    'area': area,
                    'current_score': current.get(area, 0),
                    'target_score': desired[area],
                    'gap_size': desired[area] - current.get(area, 0)
                })
        return gaps
```

```python
    def _evaluate_tools(self, tools, team_size, budget, requirements):
        """Evaluate tools based on criteria"""
        scores = {}
        
        for tool_name, tool_info in tools.items():
            score = 0
            
            # Size appropriateness
            if team_size <= 10 and 'Small teams' in tool_info.get('best_for', []):
                score += 2
            elif team_size > 10 and 'Large teams' in tool_info.get('best_for', []):
                score += 2
            
            # Feature match
            for req in requirements:
                if req.lower() in ' '.join(tool_info.get('strengths', [])).lower():
                    score += 1
            
            scores[tool_name] = score
        
        return max(scores, key=scores.get)
```
